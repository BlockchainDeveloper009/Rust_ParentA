fn main() {
    // Initialize a 3x4 matrix filled with 1s
    let matrix: Vec<Vec<i32>> = vec![vec![1; 4]; 3];

    // Access and print elements using nested loops
    for row in &matrix {
        for val in row {
            print!("{} ", val);
        }
        println!(); // Newline after each row
    }

    // Alternatively, accessing a specific element:
    println!("Element at row 2, column 3: {}", matrix[1][11]);
}
